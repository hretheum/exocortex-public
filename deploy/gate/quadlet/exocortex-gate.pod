# Publishing gate: one rootless pod for the simcheck service and the
# one-shot jobs (publisher, index, self-test, calibration).
# Host network, like the rest of this deployment: services bind to 127.0.0.1.
[Unit]
Description=Exocortex publishing gate (pod)
After=network-online.target
Wants=network-online.target

[Pod]
PodName=exocortex-gate
Network=host
# Keep the pod up when its last container exits. With podman's default
# (stop), restarting simcheck while a one-shot job was finishing stopped the
# pod, and simcheck could not start again because its pod was down.
PodmanArgs=--exit-policy=continue

[Install]
WantedBy=default.target
